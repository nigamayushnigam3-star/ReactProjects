const  App = () =>{
const [num , setNum] = useState({user:'ayush',age:20})

const btnClicked = ()=>{
  //  destructuring process 
  const newNum = {...num};
  newNum.user = 'golu'
  newNum.age = '21'
  setNum(newNum)
}
return(
  <div>
   <h1>{num.user},{num.age}</h1>
   <button onClick={btnClicked}>click me</button>

  </div>
)
}
